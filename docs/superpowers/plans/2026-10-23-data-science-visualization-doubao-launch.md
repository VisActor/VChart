```sh
python3 - <<'PY'
from pathlib import Path
from zipfile import ZipFile, ZIP_DEFLATED
import hashlib

root = Path('skills/data-science-visualization')
excluded = {'__MACOSX', '__pycache__', '.DS_Store'}
files = []
for p in sorted(root.rglob('*')):
    relative = p.relative_to(root)
    if excluded.intersection(relative.parts) or any(x.startswith('._') for x in relative.parts):
        continue
    if p.is_symlink():
        raise ValueError(f'分发包不接受符号链接: {p}')
    if p.is_file() and p.suffix != '.pyc':
        files.append(p)
out = Path('.vchart-statplots/dsv/launch/data-science-visualization.zip')
out.parent.mkdir(parents=True, exist_ok=True)
with ZipFile(out, 'x', compression=ZIP_DEFLATED) as z:
    for p in files:
        z.write(p, (Path(root.name) / p.relative_to(root)).as_posix())
with ZipFile(out) as z:
    expected = {(Path(root.name) / p.relative_to(root)).as_posix(): p.read_bytes() for p in files}
    assert set(z.namelist()) == set(expected)
    assert all(z.read(name) == content for name, content in expected.items())
    assert z.testzip() is None
print(out, len(files), hashlib.sha256(out.read_bytes()).hexdigest())
PY
```
